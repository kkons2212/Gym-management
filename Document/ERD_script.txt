// ====================================================================
// LOGICAL DATA MODEL (LDM) - GYM MANAGEMENT SYSTEM (15 TABLES)
// Visualization Tool: https://dbdiagram.io (DBML Syntax)
//
// SUMMARY OF RECENT OPTIMIZATIONS:
// 1. Kept Role table for quick UI dropdowns and referential integrity.
// 2. Member and Trainer use UserId as Shared Primary Key (1-1 with User).
// 3. Removed OperatingConfig (UC 7.2 removed; MaxFreezeDays merged into PackagePlan).
// 4. Admin lives directly in User (no separate Admin table).
// 5. Fixed list price, no discount attributes, no invoice_items.
// ====================================================================

TableGroup Authentication_Users {
  Role
  User
  Trainer
  Member
}

TableGroup Packages_CheckIn {
  PackagePlan
  MemberPackage
  PackageFreeze
  CheckInRecord
}

TableGroup Personal_Training {
  TrainerAssignment
  TrainerAvailability
  TrainingSession
  BodyMetric
  TrainerReview
}

TableGroup Finance_Billing {
  Invoice
  Refund
}

// --------------------------------------------------------------------
// 1. AUTHENTICATION & PROFILES (SHARED PRIMARY KEY 1-1)
// --------------------------------------------------------------------
Table Role {
  RoleId varchar(20) [pk]
  RoleName varchar(50) [not null]
  Description text
}

Table User {
  UserId varchar(50) [pk]
  RoleId varchar(20) [ref: > Role.RoleId, not null]
  Username varchar(50) [unique, not null]
  PasswordHash varchar(255) [not null]
  FullName varchar(100) [not null]
  Phone varchar(20) [unique, not null]
  Email varchar(100)
  is_Active boolean [not null, default: true]
  created_at timestamp [not null, default: `now()`]
}

Table Member {
  UserId varchar(50) [pk, ref: - User.UserId, note: 'PK & FK to User (Shared PK 1-1)']
  IdentityNumber varchar(20)
  DateOfBirth date
  Gender varchar(10)
  Address varchar(255)
  EmergencyContactName varchar(100)
  EmergencyContactPhone varchar(20)
  AvatarUrl varchar(500)
}

Table Trainer {
  UserId varchar(50) [pk, ref: - User.UserId, note: 'PK & FK to User (Shared PK 1-1)']
  Specialty varchar(150) [not null]
  ExperienceYears int [not null, default: 0]
  Bio text
  RatingAverage decimal(2,1) [not null, default: 5.0]
  RatingCount int [not null, default: 0]
  is_Active boolean [not null, default: true]
}

// --------------------------------------------------------------------
// 2. CATALOG
// --------------------------------------------------------------------
Table PackagePlan {
  PackagePlanId varchar(50) [pk]
  PackageName varchar(100) [not null]
  DurationDays int [not null]
  ListPrice decimal(12,2) [not null, note: 'Fixed list price']
  TotalSessions int [note: 'Session quota, null if unlimited']
  MaxFreezeDays int [not null, default: 0, note: 'Chính sách bảo lưu riêng của từng gói']
  AllowedStartHour time
  AllowedEndHour time
  has_Trainer boolean [not null, default: false]
  Description text
  is_Active boolean [not null, default: true]
}

// --------------------------------------------------------------------
// 3. MEMBER PACKAGES & CHECK-IN
// --------------------------------------------------------------------
Table MemberPackage {
  MemberPackageId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  PackagePlanId varchar(50) [ref: > PackagePlan.PackagePlanId, not null]
  QrCode varchar(100) [unique, not null, note: 'QR Code for check-in']
  StartDate date [not null]
  EndDate date [not null]
  TotalSessions int
  RemainingSessions int
  Status varchar(20) [not null, note: 'PENDING_PAYMENT, ACTIVE, FROZEN, EXPIRED, CANCELLED']
  CancellationReason text
  cancelled_at timestamp
  created_at timestamp [not null, default: `now()`]
}

Table PackageFreeze {
  PackageFreezeId varchar(50) [pk]
  MemberPackageId varchar(50) [ref: > MemberPackage.MemberPackageId, not null]
  StartDate date [not null]
  EndDate date [not null]
  DaysFrozen int [not null]
  Reason text [not null]
  created_at timestamp [not null, default: `now()`]
}

Table CheckInRecord {
  CheckInRecordId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  MemberPackageId varchar(50) [ref: > MemberPackage.MemberPackageId]
  check_in_at timestamp [not null, default: `now()`]
  Method varchar(20) [not null, note: 'QR_SCAN, MANUAL_PHONE, MANUAL_CODE']
  Status varchar(20) [not null, note: 'SUCCESS, REJECTED']
  RejectionReason varchar(255)
}

// --------------------------------------------------------------------
// 4. PERSONAL TRAINING SERVICES
// --------------------------------------------------------------------
Table TrainerAssignment {
  TrainerAssignmentId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  TrainerUserId varchar(50) [ref: > Trainer.UserId, not null]
  MemberPackageId varchar(50) [ref: > MemberPackage.MemberPackageId, not null]
  assigned_by varchar(50) [ref: > User.UserId, not null]
  AssignedDate date [not null, default: `current_date`]
  Status varchar(20) [not null, default: 'ACTIVE']
  Notes text
}

Table TrainerAvailability {
  TrainerAvailabilityId varchar(50) [pk]
  TrainerUserId varchar(50) [ref: > Trainer.UserId, not null]
  WorkDate date [not null]
  StartTime time [not null]
  EndTime time [not null]
  is_Booked boolean [not null, default: false]
}

Table TrainingSession {
  TrainingSessionId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  TrainerUserId varchar(50) [ref: > Trainer.UserId, not null]
  MemberPackageId varchar(50) [ref: > MemberPackage.MemberPackageId, not null]
  SessionDate date [not null]
  StartTime time [not null]
  EndTime time [not null]
  Status varchar(20) [not null, note: 'BOOKED, COMPLETED, MEMBER_ABSENT, TRAINER_ABSENT, CANCELLED']
  CancellationReason varchar(255)
  WorkoutNotes text
  completed_at timestamp
}

Table BodyMetric {
  BodyMetricId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  recorded_by varchar(50) [ref: > Trainer.UserId]
  RecordDate date [not null, default: `current_date`]
  WeightKg decimal(5,2) [not null]
  HeightCm decimal(5,2)
  BodyFatPercentage decimal(4,1)
  MuscleMassKg decimal(5,2)
  Bmi decimal(4,1)
  Advice text
}

Table TrainerReview {
  TrainerReviewId varchar(50) [pk]
  TrainingSessionId varchar(50) [unique, ref: - TrainingSession.TrainingSessionId, not null, note: '1-1 relation with TrainingSession']
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  TrainerUserId varchar(50) [ref: > Trainer.UserId, not null]
  Rating smallint [not null, note: '1 to 5 stars']
  Comment text
  created_at timestamp [not null, default: `now()`]
}

// --------------------------------------------------------------------
// 5. BILLING & REFUND
// --------------------------------------------------------------------
Table Invoice {
  InvoiceId varchar(50) [pk]
  MemberUserId varchar(50) [ref: > Member.UserId, not null]
  MemberPackageId varchar(50) [ref: > MemberPackage.MemberPackageId, not null]
  created_by varchar(50) [ref: > User.UserId, not null]
  Amount decimal(12,2) [not null, note: 'List price billed directly']
  PaymentMethod varchar(20) [not null, note: 'CASH, QR_TRANSFER, BANK_CARD']
  Status varchar(20) [not null, note: 'PENDING, PAID, CANCELLED, REFUNDED']
  paid_at timestamp
  created_at timestamp [not null, default: `now()`]
}

Table Refund {
  RefundId varchar(50) [pk]
  InvoiceId varchar(50) [unique, ref: - Invoice.InvoiceId, not null, note: '1-1 relation with Invoice']
  approved_by varchar(50) [ref: > User.UserId, not null]
  RefundAmount decimal(12,2) [not null]
  RefundMethod varchar(20) [not null, note: 'CASH, BANK_TRANSFER']
  Reason text [not null]
  refunded_at timestamp [not null, default: `now()`]
}
